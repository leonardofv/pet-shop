#schema Prisma

model Appointment {
  id String @id @default(cuid())
  tutorName String
  petName String
  phone String
  description String
  scheduleAt DateTime

  @@map("appointments")
}

#Comandos Prisma

pnpm add prisma && pnpm add -D @prisma/client
npx prisma init --datasource-provider postgresql
npx prisma migrate dev --name "init"
npx prisma generate
